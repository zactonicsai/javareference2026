  apiVersion: networking.istio.io/v1
  kind: VirtualService
  metadata: { name: my-app }
  spec:
    hosts: ["*"]
    gateways: ["{{ .Release.Namespace }}/{{ .Values.gateway.name }}"]
    http:
      - match: [{ uri: { prefix: /my-app/ } }]
        route: [{ destination: { host: my-app, port: { number: 8080 } } }]
